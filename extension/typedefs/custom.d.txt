-- <DRAWING> --

type DrawingBase = {
    Visible: boolean,
    ZIndex: number,
    Transparency: number,
    Color: Color3,
    Remove: (self: any) -> (),
    Destroy: (self: any) -> (),
}

export type Line = DrawingBase & {
    From: Vector2,
    To: Vector2,
    Thickness: number,
}

export type Text = DrawingBase & {
    Text: string,
    Size: number,
    Position: Vector2,
    Center: boolean,
    Outline: boolean,
    OutlineColor: Color3,
    Font: number,
    TextBounds: Vector2,
}

export type Square = DrawingBase & {
    Size: Vector2,
    Position: Vector2,
    Thickness: number,
    Filled: boolean,
}

export type Circle = DrawingBase & {
    Radius: number,
    Position: Vector2,
    Thickness: number,
    NumSides: number,
    Filled: boolean,
}

export type Triangle = DrawingBase & {
    PointA: Vector2,
    PointB: Vector2,
    PointC: Vector2,
    Thickness: number,
    Filled: boolean,
}

export type Quad = DrawingBase & {
    PointA: Vector2,
    PointB: Vector2,
    PointC: Vector2,
    PointD: Vector2,
    Thickness: number,
    Filled: boolean,
}

export type Image = DrawingBase & {
    Data: string,
    Size: Vector2,
    Position: Vector2,
    Rounding: number,
}

declare Drawing: {
    new: (("Line") -> Line)
        & (("Text") -> Text)
        & (("Square") -> Square)
        & (("Circle") -> Circle)
        & (("Triangle") -> Triangle)
        & (("Quad") -> Quad)
        & (("Image") -> Image)
        & ((kind: "Line" | "Text" | "Square" | "Circle" | "Triangle" | "Quad" | "Image") -> DrawingBase),
    Fonts: {
        UI: number,
        System: number,
        Plex: number,
        Monospace: number,
    },
}

declare function isrenderobj(object: any): boolean
declare function setrenderproperty(object: any, property: any, value: any): ()
